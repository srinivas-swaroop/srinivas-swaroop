1. **Using Integer Array (Level Order)**
2. **Using String Array with `"null"`**
3. **Using a Serialized String**
4. **Using Scanner (Recursive input)**
5. **Using Preorder + Inorder Traversal**
6. **Using Manual Insert (like BST)**

# Complete Java Program – Different Ways to Build Binary Tree

```java
import java.util.*;

/*
    Demonstrates multiple ways to build a Binary Tree

    Methods included:
    1. Build tree using Integer Array (Level Order)
    2. Build tree using String Array ("null" allowed)
    3. Build tree using Serialized String
    4. Build tree using Scanner (user input recursively)

    A preorder traversal is used to verify the tree structure.
*/

public class Main {

    // Basic Tree Node
    static class Node {
        int value;
        Node left;
        Node right;

        Node(int value){
            this.value = value;
        }
    }

    Node root;

    // ---------------------------------------------------
    // METHOD 1 : BUILD TREE USING INTEGER ARRAY
    // Level Order representation
    //
    // Example:
    // array = {1,2,3,4,5,6,7}
    //
    // Tree formed:
    //        1
    //      /   \
    //     2     3
    //    / \   / \
    //   4  5  6   7
    // ---------------------------------------------------

    Node buildTreeUsingIntegerArray(int arr[]){

        if(arr.length == 0)
            return null;

        Node root = new Node(arr[0]);

        Queue<Node> queue = new LinkedList<>();
        queue.add(root);

        int i = 1;

        while(i < arr.length){

            Node current = queue.poll();

            // create left child
            current.left = new Node(arr[i++]);
            queue.add(current.left);

            if(i < arr.length){
                // create right child
                current.right = new Node(arr[i++]);
                queue.add(current.right);
            }
        }

        return root;
    }

    // ---------------------------------------------------
    // METHOD 2 : BUILD TREE USING STRING ARRAY
    // "null" means no node
    //
    // Example:
    // {"1","2","3","null","5"}
    //
    // Tree:
    //        1
    //       / \
    //      2   3
    //       \
    //        5
    // ---------------------------------------------------

    Node buildTreeUsingStringArray(String arr[]){

        if(arr.length == 0 || arr[0].equals("null"))
            return null;

        Node root = new Node(Integer.parseInt(arr[0]));

        Queue<Node> queue = new LinkedList<>();
        queue.add(root);

        int i = 1;

        while(!queue.isEmpty() && i < arr.length){

            Node current = queue.poll();

            if(!arr[i].equals("null")){
                current.left = new Node(Integer.parseInt(arr[i]));
                queue.add(current.left);
            }
            i++;

            if(i < arr.length && !arr[i].equals("null")){
                current.right = new Node(Integer.parseInt(arr[i]));
                queue.add(current.right);
            }
            i++;
        }

        return root;
    }

    // ---------------------------------------------------
    // METHOD 3 : BUILD TREE USING SERIALIZED STRING
    //
    // Input Example:
    // "1 2 3 null 5"
    //
    // First split string then reuse previous method
    // ---------------------------------------------------

    Node buildTreeUsingString(String s){

        String arr[] = s.split(" ");
        return buildTreeUsingStringArray(arr);
    }

    // ---------------------------------------------------
    // METHOD 4 : BUILD TREE USING SCANNER (RECURSIVE)
    //
    // User enters -1 for NULL node
    //
    // Example input:
    // 1
    // 2
    // -1
    // -1
    // 3
    // -1
    // -1
    //
    // Tree:
    //      1
    //     / \
    //    2   3
    // ---------------------------------------------------

    Node buildTreeUsingScanner(Scanner sc){

        System.out.print("Enter value (-1 for null): ");

        int val = sc.nextInt();

        if(val == -1)
            return null;

        Node node = new Node(val);

        System.out.println("Left of " + val);
        node.left = buildTreeUsingScanner(sc);

        System.out.println("Right of " + val);
        node.right = buildTreeUsingScanner(sc);

        return node;
    }

    // ---------------------------------------------------
    // PREORDER TRAVERSAL
    // Used to verify the tree
    // ---------------------------------------------------

    void preorder(Node root){

        if(root == null)
            return;

        System.out.print(root.value + " ");

        preorder(root.left);
        preorder(root.right);
    }

    // ---------------------------------------------------
    // MAIN FUNCTION
    // ---------------------------------------------------

    public static void main(String[] args) {

        Main tree = new Main();

        // =================================================
        // METHOD 1 : INTEGER ARRAY
        // =================================================

        System.out.println("---- Build Tree Using Integer Array ----");

        int arr[] = {1,2,3,4,5,6,7};

        Node root1 = tree.buildTreeUsingIntegerArray(arr);

        System.out.print("Preorder Traversal: ");
        tree.preorder(root1);
        System.out.println();


        // =================================================
        // METHOD 2 : STRING ARRAY
        // =================================================

        System.out.println("\n---- Build Tree Using String Array ----");

        String arr2[] = {"1","2","3","null","5"};

        Node root2 = tree.buildTreeUsingStringArray(arr2);

        System.out.print("Preorder Traversal: ");
        tree.preorder(root2);
        System.out.println();


        // =================================================
        // METHOD 3 : SERIALIZED STRING
        // =================================================

        System.out.println("\n---- Build Tree Using Serialized String ----");

        String s = "1 2 3 null 5";

        Node root3 = tree.buildTreeUsingString(s);

        System.out.print("Preorder Traversal: ");
        tree.preorder(root3);
        System.out.println();


        // =================================================
        // METHOD 4 : SCANNER INPUT
        // =================================================

        System.out.println("\n---- Build Tree Using Scanner ----");

        Scanner sc = new Scanner(System.in);

        Node root4 = tree.buildTreeUsingScanner(sc);

        System.out.print("Preorder Traversal: ");
        tree.preorder(root4);

    }
}
```
